-- ============================================
local function makeDraggable(guiObject, dragHandle)
	dragHandle = dragHandle or guiObject
	local dragging = false
	local dragStart, startPos

	dragHandle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = guiObject.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	dragHandle.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			guiObject.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y
			)
		end
	end)
end

-- FPS BOOST LOGIC

local CULL_DISTANCE = 450
local CULL_INTERVAL = 0.4

local trackedParts = {}
local cullRunning = false
local savedStreamingEnabled = Workspace.StreamingEnabled

local function isExcluded(obj)
	local charFolder = obj:FindFirstAncestorOfClass("Model")
	if charFolder and Players:GetPlayerFromCharacter(charFolder) then
		return true
	end
	return false
end

local function collectParts()
	trackedParts = {}
	for _, obj in ipairs(Workspace:GetDescendants()) do
		if obj:IsA("BasePart") and not isExcluded(obj) then
			table.insert(trackedParts, obj)
		end
	end
end

local addedConn, removedConn, cullLoopThread

local savedMaterials = {}
local materialFlattenOn = false
local lowTextureMode = false

local function startDistanceCulling()
	if cullRunning then return end
	cullRunning = true
	collectParts()

	addedConn = Workspace.DescendantAdded:Connect(function(obj)
		if cullRunning and obj:IsA("BasePart") and not isExcluded(obj) then
			table.insert(trackedParts, obj)
			if materialFlattenOn and obj.Material ~= Enum.Material.SmoothPlastic then
				savedMaterials[obj] = obj.Material
				obj.Material = Enum.Material.SmoothPlastic
			end
			if lowTextureMode then
				obj.Color = Color3.fromRGB(180, 180, 180)
			end
		end
	end)
	removedConn = Workspace.DescendantRemoving:Connect(function(obj)
		if obj:IsA("BasePart") then
			for i = #trackedParts, 1, -1 do
				if trackedParts[i] == obj then
					table.remove(trackedParts, i)
					break
				end
			end
		end
	end)

	cullLoopThread = task.spawn(function()
		while cullRunning do
			local camPos = camera.CFrame.Position
			for i = 1, #trackedParts do
				local part = trackedParts[i]
				if part and part.Parent then
					local dist = (part.Position - camPos).Magnitude
					if dist > CULL_DISTANCE then
						part.LocalTransparencyModifier = 1
					else
						part.LocalTransparencyModifier = 0
					end
				end
			end
			task.wait(CULL_INTERVAL)
		end
	end)
end

local function stopDistanceCulling()
	cullRunning = false
	if addedConn then addedConn:Disconnect(); addedConn = nil end
	if removedConn then removedConn:Disconnect(); removedConn = nil end
	for _, part in ipairs(trackedParts) do
		if part and part.Parent then
			part.LocalTransparencyModifier = 0
		end
	end
	trackedParts = {}
end

local function flattenMaterials(instance)
	for _, obj in ipairs(instance:GetDescendants()) do
		if obj:IsA("BasePart") and obj.Material ~= Enum.Material.SmoothPlastic then
			savedMaterials[obj] = obj.Material
			obj.Material = Enum.Material.SmoothPlastic
		end
	end
end

local function restoreMaterials()
	for part, mat in pairs(savedMaterials) do
		if part and part.Parent then
			part.Material = mat
		end
	end
	savedMaterials = {}
end

local function applyLowTexture(instance)
	for _, obj in ipairs(instance:GetDescendants()) do
		if obj:IsA("BasePart") then
			obj.Color = Color3.fromRGB(180, 180, 180)
			obj.Material = Enum.Material.SmoothPlastic
		end
	end
end

local function restoreTexture()
	for part, mat in pairs(savedMaterials) do
		if part and part.Parent then
			part.Material = mat
		end
	end
	savedMaterials = {}
end

local boosted = false
local savedSky = nil
local savedTechnology = Lighting.Technology

local function fpsBoost()
	for _, effect in ipairs(Lighting:GetChildren()) do
		if effect:IsA("BlurEffect") or effect:IsA("DepthOfFieldEffect") or effect:IsA("BloomEffect")
			or effect:IsA("SunRaysEffect") or effect:IsA("ColorCorrectionEffect")
			or effect:IsA("Atmosphere") or effect.ClassName == "Clouds" then
			effect:Destroy()
		end
	end

	local sky = Lighting:FindFirstChildOfClass("Sky")
	if sky then
		savedSky = sky:Clone()
		sky:Destroy()
	end

	Lighting.Technology = Enum.Technology.Compatibility
	Lighting.GlobalShadows = false

	local terrain = Workspace:FindFirstChildOfClass("Terrain")
	if terrain then
		terrain.Decoration = false
		terrain.WaterWaveSize = 0
		terrain.WaterWaveSpeed = 0
		terrain.WaterReflectance = 0
	end

	local function nukeHeavyEffects(instance)
		for _, obj in ipairs(instance:GetDescendants()) do
			if obj:IsA("ParticleEmitter") or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles")
				or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Decal") or obj:IsA("Texture")
				or obj:IsA("Highlight") or obj:IsA("SelectionBox") or obj:IsA("SurfaceAppearance") then
				obj:Destroy()
			elseif obj:IsA("BasePart") then
				obj.CastShadow = false
			end
		end
	end
	nukeHeavyEffects(Workspace)

	pcall(function()
		Workspace.StreamingEnabled = true
		Workspace.StreamingMinRadius = 128
		Workspace.StreamingTargetRadius = 256
	end)

	startDistanceCulling()

	materialFlattenOn = true
	flattenMaterials(Workspace)

	pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
	pcall(function() settings().Rendering.MeshPartHeadsAndAccessories = false end)

	collectgarbage("collect")
	task.spawn(function()
		while boosted do
			task.wait(30)
			collectgarbage("collect")
		end
	end)

	boosted = true
	fpsBtn.Text = "DISABLE FPS BOOST ✓"
end

local function fpsUnboost()
	if savedSky then
		savedSky.Parent = Lighting
		savedSky = nil
	end
	Lighting.Technology = savedTechnology
	Lighting.GlobalShadows = true

	local terrain = Workspace:FindFirstChildOfClass("Terrain")
	if terrain then terrain.Decoration = true end

	pcall(function() Workspace.StreamingEnabled = savedStreamingEnabled end)

	stopDistanceCulling()

	materialFlattenOn = false
	restoreMaterials()

	if lowTextureMode then
		restoreTexture()
		lowTextureMode = false
		textureBtn.Text = "LOW TEXTURE: OFF"
	end

	boosted = false
	fpsBtn.Text = "ENABLE FPS BOOST"
end

fpsBtn.MouseButton1Click:Connect(function()
	if boosted then fpsUnboost() else fpsBoost() end
end)

textureBtn.MouseButton1Click:Connect(function()
	lowTextureMode = not lowTextureMode
	if lowTextureMode then
		applyLowTexture(Workspace)
		textureBtn.Text = "LOW TEXTURE: ON"
	else
		restoreTexture()
		textureBtn.Text = "LOW TEXTURE: OFF"
	end
end)

Lighting.DescendantAdded:Connect(function(effect)

		or effect:IsA("SunRaysEffect") or effect:IsA("ColorCorrectionEffect")

		effect:Destroy()
	end
end)

-- AIMBOT MỚI (CÓ FOV RING)

local teamCheck = false
local fov = 150
local smoothing = 0.3
local aimbotEnabled = false
local aimbotLoop = nil

-- FOV Ring
local FOVring = Drawing.new("Circle")
FOVring.Visible = false
FOVring.Thickness = 1.5
FOVring.Radius = fov
FOVring.Transparency = 1
FOVring.Color = Color3.fromRGB(255, 128, 128)
FOVring.Position = camera.ViewportSize/2

local function getClosest(cframe)
	local ray = Ray.new(cframe.Position, cframe.LookVector).Unit
	
	local target = nil
	local mag = math.huge
	
	for i,v in pairs(Players:GetPlayers()) do
		if v.Character and v.Character:FindFirstChild("Head") and v.Character:FindFirstChild("Humanoid") and v.Character:FindFirstChild("HumanoidRootPart") and v ~= player and (v.Team ~= player.Team or (not teamCheck)) then
			local magBuf = (v.Character.Head.Position - ray:ClosestPoint(v.Character.Head.Position)).Magnitude
			
			if magBuf < mag then
				mag = magBuf
				target = v
			end
		end
	end
	
	return target
end

local function startAimbot()
	if aimbotLoop then return end
	FOVring.Visible = true
	
	aimbotLoop = RunService.RenderStepped:Connect(function()
		local pressed = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
		local localPlay = player.Character
		local cam = camera
		local zz = camera.ViewportSize/2
		
		FOVring.Position = camera.ViewportSize/2
		
		if pressed and localPlay then
			local curTar = getClosest(cam.CFrame)
			if curTar and curTar.Character and curTar.Character:FindFirstChild("Head") then
				local ssHeadPoint = cam:WorldToScreenPoint(curTar.Character.Head.Position)
				ssHeadPoint = Vector2.new(ssHeadPoint.X, ssHeadPoint.Y)
				if (ssHeadPoint - zz).Magnitude < fov then
					camera.CFrame = camera.CFrame:Lerp(CFrame.new(cam.CFrame.Position, curTar.Character.Head.Position), smoothing)
				end
			end
		end
	end)
end

local function stopAimbot()
	if aimbotLoop then
		aimbotLoop:Disconnect()
		aimbotLoop = nil
	end
	FOVring.Visible = false
end

-- ============================================
-- ESP BOX (CÓ BO GÓC)
-- ============================================
local espEnabled = false
local espBoxes = {}
local espConnection = nil
local cornerLength = 10 -- Độ dài của góc bo

local function createESPBox(targetPlayer)
	local boxData = {
		-- 4 đường line cho box
		topLine = Drawing.new("Line"),
		bottomLine = Drawing.new("Line"),
		leftLine = Drawing.new("Line"),
		rightLine = Drawing.new("Line"),
		-- 4 góc bo
		topLeftCorner = Drawing.new("Line"),
		topRightCorner = Drawing.new("Line"),
		bottomLeftCorner = Drawing.new("Line"),
		bottomRightCorner = Drawing.new("Line"),
	}
	
	-- Cấu hình các line
	for _, line in pairs(boxData) do
		line.Thickness = 2
		line.Color = Color3.fromRGB(255, 0, 0)
		line.Transparency = 1
		line.Visible = false
	end
	
	espBoxes[targetPlayer] = boxData
	return boxData
end

local function updateESPBox(targetPlayer, boxData)
	local targetChar = targetPlayer.Character
	if targetChar and targetPlayer ~= player then
		local targetHumanoid = targetChar:FindFirstChildOfClass("Humanoid")
		local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
		if targetHumanoid and targetRoot and targetHumanoid.Health > 0 then
			local head = targetChar:FindFirstChild("Head")
			if head then
				local headPos = camera:WorldToViewportPoint(head.Position)
				local rootPos = camera:WorldToViewportPoint(targetRoot.Position)
				
				if headPos.Z > 0 and rootPos.Z > 0 then
					local height = (headPos.Y - rootPos.Y) * 2
					local width = height * 0.6
					
					local x = rootPos.X
					local y = rootPos.Y
					local top = y - height/2
					local bottom = y + height/2
					local left = x - width/2
					local right = x + width/2
					
					-- Vẽ box với các góc bo
					-- Top line (bỏ phần góc)
					boxData.topLine.From = Vector2.new(left + cornerLength, top)
					boxData.topLine.To = Vector2.new(right - cornerLength, top)
					boxData.topLine.Visible = true
					
					-- Bottom line (bỏ phần góc)
					boxData.bottomLine.From = Vector2.new(left + cornerLength, bottom)
					boxData.bottomLine.To = Vector2.new(right - cornerLength, bottom)
					boxData.bottomLine.Visible = true
					
					-- Left line (bỏ phần góc)
					boxData.leftLine.From = Vector2.new(left, top + cornerLength)
					boxData.leftLine.To = Vector2.new(left, bottom - cornerLength)
					boxData.leftLine.Visible = true
					
					-- Right line (bỏ phần góc)
					boxData.rightLine.From = Vector2.new(right, top + cornerLength)
					boxData.rightLine.To = Vector2.new(right, bottom - cornerLength)
					boxData.rightLine.Visible = true
					
					-- Top-left corner
					boxData.topLeftCorner.From = Vector2.new(left, top + cornerLength)
					boxData.topLeftCorner.To = Vector2.new(left + cornerLength, top)
					boxData.topLeftCorner.Visible = true
					
					-- Top-right corner
					boxData.topRightCorner.From = Vector2.new(right - cornerLength, top)
					boxData.topRightCorner.To = Vector2.new(right, top + cornerLength)
					boxData.topRightCorner.Visible = true
					
					-- Bottom-left corner
					boxData.bottomLeftCorner.From = Vector2.new(left, bottom - cornerLength)
					boxData.bottomLeftCorner.To = Vector2.new(left + cornerLength, bottom)
					boxData.bottomLeftCorner.Visible = true
					
					-- Bottom-right corner
					boxData.bottomRightCorner.From = Vector2.new(right - cornerLength, bottom)
					boxData.bottomRightCorner.To = Vector2.new(right, bottom - cornerLength)
					boxData.bottomRightCorner.Visible = true
				else
					-- Ẩn tất cả nếu ngoài màn hình
					for _, line in pairs(boxData) do
						line.Visible = false
					end
				end
			end
		else
			-- Ẩn nếu player chết hoặc không có character
			for _, line in pairs(boxData) do
				line.Visible = false
			end
		end
	else
		-- Ẩn nếu là player hiện tại
		for _, line in pairs(boxData) do
			line.Visible = false
		end
	end
end

local function startESP()
	if espConnection then return end
	
	-- Tạo box cho tất cả player hiện tại
	for _, targetPlayer in ipairs(Players:GetPlayers()) do
		if targetPlayer ~= player and not espBoxes[targetPlayer] then
			createESPBox(targetPlayer)
		end
	end
	
	-- Theo dõi player mới
	Players.PlayerAdded:Connect(function(targetPlayer)
		task.wait(1)
		if espEnabled and targetPlayer ~= player and not espBoxes[targetPlayer] then
			createESPBox(targetPlayer)
		end
	end)
	
	Players.PlayerRemoving:Connect(function(targetPlayer)
		if espBoxes[targetPlayer] then
			for _, line in pairs(espBoxes[targetPlayer]) do
				line:Remove()
			end
			espBoxes[targetPlayer] = nil
		end
	end)
	
	espConnection = RunService.RenderStepped:Connect(function()
		if espEnabled then
			for targetPlayer, boxData in pairs(espBoxes) do
				updateESPBox(targetPlayer, boxData)
			end
		end
	end)
end

local function stopESP()
	if espConnection then
		espConnection:Disconnect()
		espConnection = nil
	end
	
	-- Ẩn tất cả box
	for targetPlayer, boxData in pairs(espBoxes) do
		for _, line in pairs(boxData) do
			line.Visible = false
		end
	end
end

espBtn.MouseButton1Click:Connect(function()
	espEnabled = not espEnabled
	if espEnabled then
		startESP()
		espBtn.Text = "ESP: ON"
	else
		stopESP()
		espBtn.Text = "ESP: OFF"
	end
end)
